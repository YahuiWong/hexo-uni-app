import mermaid from 'mermaid';

onMounted(() => {
  mermaid.initialize({
    startOnLoad: false,
    theme: 'dark'  // 黑客风格
  });

  const elements = document.querySelectorAll('.mermaid');
  elements.forEach((el, index) => {
    mermaid.render(`mermaid-${index}`, el.textContent, (svg) => {
      el.innerHTML = svg;
    });

});

import katex from 'katex';

// 渲染行内公式
function renderMath() {
  document.querySelectorAll('.math-inline').forEach(el => {
    const latex = el.textContent.replace(/^\$|\$$/g, '');
    katex.render(latex, el, { throwOnError: false });
  });

  // 渲染块级公式
  document.querySelectorAll('.math-block').forEach(el => {
    const latex = el.textContent.replace(/^\$\$|\$\$$/g, '');
    katex.render(latex, el, {
      throwOnError: false,
      displayMode: true
    });
  });
}

## 🔥 推荐实现：一体化方案

结合以上方案，创建完整的渲染系统：

### 1. 代码高亮 - Prism.js + 黑客主题

```typescript
import Prism from 'prismjs';
import 'prismjs/themes/prism-okaidia.css';  // 黑客风格主题

// 支持的语言
import 'prismjs/components/prism-javascript';
import 'prismjs/components/prism-typescript';
import 'prismjs/components/prism-python';
import 'prismjs/components/prism-java';
import 'prismjs/components/prism-css';
import 'prismjs/components/prism-markup';

renderer.code = function (token: any): string {
  const code = token.text || '';
  const lang = (token.lang || '').trim().toLowerCase();

  if (Prism.languages[lang]) {
    const highlighted = Prism.highlight(code, Prism.languages[lang], lang);
    return `<pre class="language-${lang}"><code>${highlighted}</code></pre>`;
  }

  return `<pre><code>${escapeHtml(code)}</code></pre>`;
};
```

### 2. Mermaid 渲染

```typescript
import mermaid from 'mermaid';

onMounted(async () => {
  mermaid.initialize({
    startOnLoad: false,
    theme: 'dark'
  });

  const mermaidElements = document.querySelectorAll('.mermaid');
  for (let i = 0; i < mermaidElements.length; i++) {
    try {
      const { svg } = await mermaid.render(
        `mermaid-${i}`,
        mermaidElements[i].textContent
      );
      mermaidElements[i].innerHTML = svg;
    } catch (e) {
      console.error('Mermaid render error:', e);
    }
  }
});
```
